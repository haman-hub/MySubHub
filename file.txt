// ================== DATABASE HELPERS ==================
async function dbSelect(table: string, filters: Record<string, unknown> = {}, select = '*') {
  let query = supabase.from(table).select(select);
  for (const [key, value] of Object.entries(filters)) {
    if (value !== undefined && value !== null) {
      query = query.eq(key, value);
    }
  }
  const { data, error } = await query;
  if (error) throw new Error(`DB error [${table}]: ${error.message}`);
  return data || [];
}

async function dbInsert(table: string, body: Record<string, unknown>) {
  const { data, error } = await supabase.from(table).insert(body).select();
  if (error) throw new Error(`DB insert error [${table}]: ${error.message}`);
  return data;
}

async function dbUpdate(table: string, filters: Record<string, unknown>, body: Record<string, unknown>) {
  let query = supabase.from(table).update(body);
  for (const [key, value] of Object.entries(filters)) {
    if (value !== undefined && value !== null) {
      query = query.eq(key, value);
    }
  }
  const { data, error } = await query.select();
  if (error) throw new Error(`DB update error [${table}]: ${error.message}`);
  return data;
}

async function dbDelete(table: string, filters: Record<string, unknown>) {
  let query = supabase.from(table).delete();
  for (const [key, value] of Object.entries(filters)) {
    if (value !== undefined && value !== null) {
      query = query.eq(key, value);
    }
  }
  const { error } = await query;
  if (error) throw new Error(`DB delete error [${table}]: ${error.message}`);
  return true;
}

async function dbRpc(fn: string, params: Record<string, unknown> = {}) {
  const { data, error } = await supabase.rpc(fn, params);
  if (error) throw new Error(`RPC error [${fn}]: ${error.message}`);
  return data;
}

// ================== UTILITIES ==================
function isValidUUID(str: string): boolean {
  return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(str);
}

function escapeHtml(str: string): string {
  if (!str) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#x27;');
}

function generateReferralCode(): string {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  let code = '';
  for (let i = 0; i < 8; i++) code += chars.charAt(Math.floor(Math.random() * chars.length));
  return code;
}

// ================== CORS ==================
const ALLOWED_ORIGINS = ["https://haman-hub.github.io", "https://web.telegram.org", "https://t.me"];

function getCorsHeaders(origin: string | null) {
  const headers: Record<string, string> = {
    "Access-Control-Allow-Methods": "GET, POST, PUT, PATCH, DELETE, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type, x-telegram-initdata, Authorization, x-cron-secret",
    "Access-Control-Max-Age": "86400",
    "Vary": "Origin",
  };
  // Exact origin matching avoids trusting hostile look-alike origins.
  if (origin && ALLOWED_ORIGINS.includes(origin)) headers["Access-Control-Allow-Origin"] = origin;
  return headers;
}

function jsonResponse(payload: unknown, status = 200, origin: string | null = null) {
  return new Response(JSON.stringify(payload), {
    status,
    headers: { "Content-Type": "application/json", ...getCorsHeaders(origin) },
  });
}

// ================== RATE LIMITING ==================
const rateLimitMap = new Map<string, { count: number; resetAt: number }>();

setInterval(() => {
  const now = Date.now();
  for (const [key, entry] of rateLimitMap) {
    if (now > entry.resetAt) rateLimitMap.delete(key);
  }
}, 300_000);

function checkRateLimit(userId: string): boolean {
  const now = Date.now();
  const entry = rateLimitMap.get(userId);
  if (!entry || now > entry.resetAt) {
    rateLimitMap.set(userId, { count: 1, resetAt: now + 60000 });
    return true;
  }
  if (entry.count >= 30) return false;
  entry.count++;
  return true;
}

async function telegramApi(method: string, body: Record<string, unknown> = {}) {
  if (!BOT_TOKEN) throw new Error("BOT_TOKEN not configured");
  console.log(`📤 Telegram API: ${method}`);

  return data;
}

// ================== INITDATA VALIDATION ==================
async function validateInitData(initData: string): Promise<Record<string, unknown> | null> {
  try {
    if (!initData || !BOT_TOKEN) return null;
    const params = new URLSearchParams(initData);
    const authDate = parseInt(params.get("auth_date") || "0", 10);
    const now = Math.floor(Date.now() / 1000);
    if (!Number.isFinite(authDate) || authDate <= 0 || authDate > now + 30 || now - authDate > 86400) return null;

    const secretKey = await crypto.subtle.importKey(
      "raw", new TextEncoder().encode("WebAppData"),
      { name: "HMAC", hash: "SHA-256" }, false, ["sign"]
    );
    const key = await crypto.subtle.sign("HMAC", secretKey, new TextEncoder().encode(BOT_TOKEN));
    const hash = params.get("hash");
    params.delete("hash");
    const sorted = new URLSearchParams();
    Array.from(params.keys()).sort().forEach(k => sorted.set(k, params.get(k)!));
    const dataCheckString = Array.from(sorted.entries()).map(([k, v]) => `${k}=${v}`).join("\n");
    const hmac = await crypto.subtle.sign(
      "HMAC",
      await crypto.subtle.importKey("raw", key, { name: "HMAC", hash: "SHA-256" }, false, ["sign"]),
      new TextEncoder().encode(dataCheckString)
    );
    const calculated = Array.from(new Uint8Array(hmac)).map(b => b.toString(16).padStart(2, "0")).join("");
    if (calculated !== hash) return null;

    const userJson = params.get("user");
    if (!userJson) return null;
    return JSON.parse(userJson);
  } catch (e) {
    console.error("InitData validation error:", e);
    return null;
  }
}

// ================== TON PAYMENT VERIFICATION ==================
function normalizeTonAddress(value: string): string {
  try {
    return Address.parse(value).toRawString().toLowerCase();
  } catch {
    return String(value || "").trim().toLowerCase();
  }
}

function makeTonCommentPayload(comment: string): string {
  // TON text-comment payload: uint32 opcode 0 followed by UTF-8 text.
  const cell = beginCell().storeUint(0, 32).storeStringTail(comment).endCell();
  const boc = new Uint8Array(cell.toBoc() as Uint8Array);
  return btoa(Array.from(boc, (byte: number) => String.fromCharCode(byte)).join(""));
}

async function verifyTonPayment(
  walletAddress: string,
  expectedAmountNano: bigint,
  comment: string,
  createdAfter: number,
): Promise<{ success: boolean; txHash?: string; error?: string }> {
  try {
    if (!TONCENTER_API_KEY) return { success: false, error: "TON Center API key is not configured" };
    if (expectedAmountNano <= 0n) return { success: false, error: "Invalid expected amount" };

    const endpoint = new URL(`${TONCENTER_BASE}/getTransactions`);
    endpoint.searchParams.set("address", walletAddress);
    endpoint.searchParams.set("limit", "100");
    const response = await fetch(endpoint.toString(), {
      headers: { "X-API-Key": TONCENTER_API_KEY },
    });
    if (!response.ok) return { success: false, error: "TON Center transaction lookup failed" };

    const data = await response.json();
    if (!data?.ok || !Array.isArray(data.result)) {
      return { success: false, error: "TON Center returned an invalid transaction response" };
    }

    const expectedDestination = normalizeTonAddress(walletAddress);
    for (const tx of data.result) {
      if (!tx?.utime || Number(tx.utime) < createdAfter) continue;
      if (tx.aborted === true || tx.description?.aborted === true ||
          tx.compute_phase?.success === false || tx.action_phase?.success === false) continue;

      const incoming = tx.in_msg;
      if (!incoming?.value || !incoming?.destination || typeof incoming.message !== "string") continue;
      if (incoming.bounced === true) continue;
      if (normalizeTonAddress(String(incoming.destination)) !== expectedDestination) continue;

      let receivedNano: bigint;
      try { receivedNano = BigInt(String(incoming.value)); } catch { continue; }

      // Exact amount + per-intent comment prevents amount-only matching and basic replay.
      if (receivedNano !== expectedAmountNano || incoming.message !== comment) continue;

      const txHash = tx.transaction_id?.hash;
      if (!txHash || typeof txHash !== "string") continue;
      const duplicate = await dbSelect("subscriptions", { transaction_hash: txHash });
      if (duplicate.length > 0) continue;

      return { success: true, txHash };
    }
    return {
      success: false,
      error: "No confirmed transfer matching recipient, exact amount and payment reference was found",
    };
  } catch (e: unknown) {
    return {
      success: false,
      error: `Payment verification error: ${e instanceof Error ? e.message : "Unknown"}`,
    };
  }
}

async function verifyTonPayout(
  transactionHash: string,
  destination: string,
  expectedAmountNano: bigint,
  approvedAfter: number,
): Promise<boolean> {
  try {
    if (!TONCENTER_API_KEY || !ADMIN_TON_WALLET || !transactionHash) return false;

    const endpoint = new URL(`${TONCENTER_BASE}/getTransactions`);
    endpoint.searchParams.set("address", ADMIN_TON_WALLET);
    endpoint.searchParams.set("limit", "100");
    const response = await fetch(endpoint.toString(), {
      headers: { "X-API-Key": TONCENTER_API_KEY },
    });
    if (!response.ok) return false;

    const data = await response.json();
    if (!data?.ok || !Array.isArray(data.result)) return false;
    const expectedDestination = normalizeTonAddress(destination);

    for (const tx of data.result) {
      if (tx?.transaction_id?.hash !== transactionHash) continue;
      if (Number(tx.utime || 0) < approvedAfter || tx.aborted === true ||
          tx.description?.aborted === true || tx.compute_phase?.success === false ||
          tx.action_phase?.success === false) return false;

      for (const outgoing of (tx.out_msgs || [])) {
        if (!outgoing?.destination || !outgoing?.value) continue;
        if (normalizeTonAddress(String(outgoing.destination)) !== expectedDestination) continue;
        try {
          if (BigInt(String(outgoing.value)) === expectedAmountNano) return true;
        } catch {
          // Ignore malformed message values.
        }
      }
      return false;
    }
    return false;
  } catch (e) {
    console.error("Payout verification error:", e);
    return false;
  }
}

// ================== BOT ADMIN VERIFICATION ==================
async function verifyBotIsAdmin(channelInviteLink: string): Promise<{ success: boolean; error?: string }> {
  try {
    const match = channelInviteLink.match(/t\.me\/\+?([a-zA-Z0-9_]+)/);
    if (!match) {
      return { success: false, error: "Invalid invite link format" };
    }
    
    const channelIdentifier = match[1];
    const chatInfo = await telegramApi("getChat", { chat_id: channelIdentifier });
    
    if (!chatInfo.ok) {
      return { success: false, error: "Could not access channel. Make sure the invite link is correct and the bot can access it." };
    }
    
    const chatId = chatInfo.result.id;
    const botMember = await telegramApi("getChatMember", { 
      chat_id: chatId, 
      user_id: (await telegramApi("getMe")).result.id 
    });
    
    if (!botMember.ok) {
      return { success: false, error: "Bot is not a member of this channel" };
    }
    
    const status = botMember.result.status;
    
    if (status !== "administrator" && status !== "creator") {
      return { 
        success: false, 
        error: "Bot must be added as administrator to this channel first. Please add the bot as admin and try again." 
      };
    }
    
    return { success: true };
  } catch (e: unknown) {
    return { 
      success: false, 
      error: "Failed to verify bot admin status. Please ensure the bot has been added as administrator to your channel." 
    };
  }
}

// ================== SUBSCRIPTION EXPIRATION ==================
async function expireSubscription(subId: string) {
  try {
    const subs = await dbSelect("subscriptions", { id: subId });
    if (subs.length === 0) return { success: false, error: "Subscription not found" };
    
    const sub = subs[0];
    const userId = sub.user_id as number;
    const channelId = sub.channel_id as string;
    
    const channels = await dbSelect("channels", { id: channelId });
    if (channels.length === 0) return { success: false, error: "Channel not found" };
    
    const channel = channels[0];
    const channelChatId = channel.channel_chat_id as number;
    
    await dbUpdate("subscriptions", { id: subId }, { 
      status: 'expired',
      updated_at: new Date().toISOString()
    });
    
    if (channelChatId) {
      try {
        await telegramApi("banChatMember", {
          chat_id: channelChatId,
          user_id: userId,
          revoke_messages: false
        });
        console.log(`✅ User ${userId} removed from channel ${channelChatId}`);
      } catch (e) {
        console.error(`❌ Failed to remove user:`, e);
      }
    }
    
    try {
      await telegramApi("sendMessage", {
        chat_id: userId,
        text: `⚠️ Your subscription to "${escapeHtml(channel.channel_name as string)}" has expired.\n\nRenew it from the app!`,
        reply_markup: {
          inline_keyboard: [[
            { text: "🔄 Renew", web_app: { url: `${WEBAPP_URL}?startapp=${channelId}` } }
          ]]
        }
      });
    } catch (e) {
      console.error("Failed to notify user:", e);
    }
    
    return { success: true, userId, channelId };
  } catch (e: unknown) {
    return { success: false, error: e instanceof Error ? e.message : 'Unknown' };
  }
}

            inline_keyboard: [[{ text: "⚙️ Configure Channel", web_app: { url: `${WEBAPP_URL}?startapp=owner` } }]]
          }
        });
        
        console.log(`✅ Channel registered: ${chat.title} (${channelId})`);
      } catch (e) {
        console.error("Channel registration error:", e);
      }
    }
  }
}

// ================== API HANDLER ==================
async function apiHandler(path: string, method: string, headers: Headers, body: Record<string, unknown>, origin: string | null) {
  const initData = headers.get("x-telegram-initdata") || "";
  const tgUser = await validateInitData(initData);
  
  if (!tgUser || !tgUser.id) {
    return jsonResponse({ error: "Unauthorized: Invalid Telegram data" }, 401, origin);
  }
  
  const telegramId = tgUser.id as number;
  
  let user: Record<string, unknown>;
  try {
    const users = await dbSelect("users", { telegram_id: telegramId });
    if (users.length === 0) {
      const referralCode = generateReferralCode();
      await dbInsert("users", {
        telegram_id: telegramId,
        first_name: tgUser.first_name,
        last_name: tgUser.last_name,
        username: tgUser.username,
        language_code: (tgUser.language_code as string) || 'en',
        referral_code: referralCode,
      });
      user = { telegram_id: telegramId, first_name: tgUser.first_name, referral_code: referralCode, is_banned: false, role: 'user' };
    } else {
      user = users[0];
    }
  } catch (e) {
    console.error("User auth error:", e);
    return jsonResponse({ error: "Authentication failed" }, 500, origin);
  }
  
  if (user.is_banned) {
    return jsonResponse({ error: "Account suspended" }, 403, origin);
  }
  
  const userIdStr = telegramId.toString();
  if (!checkRateLimit(userIdStr)) {
    return jsonResponse({ error: "Too many requests" }, 429, origin);
  }
  
  const isAdmin = userIdStr === ADMIN_TELEGRAM_ID;
  
  if (path === "/api/auth/validate" && (method === "GET" || method === "POST")) {
    return jsonResponse({
      user: {
        telegram_id: user.telegram_id,
        first_name: user.first_name,
        last_name: user.last_name,
        username: user.username,
        wallet_address: user.wallet_address,
        role: user.role,
        referral_code: user.referral_code,
      },
      isAdmin,
    }, 200, origin);
  }
  
  if (path === "/api/auth/wallet" && method === "POST") {
    const { wallet_address } = body;
    if (!wallet_address || typeof wallet_address !== "string" || wallet_address.length > 128) {
      return jsonResponse({ error: "Invalid wallet address" }, 400, origin);
    }
    try {
      const canonical = Address.parse(wallet_address).toString({ urlSafe: true, bounceable: false });
      await dbUpdate("users", { telegram_id: telegramId }, { wallet_address: canonical });
      return jsonResponse({ success: true, wallet_address: canonical }, 200, origin);
    } catch {
      return jsonResponse({ error: "Invalid TON wallet address" }, 400, origin);
    }
  }
  
  // ================== CHANNEL DISCOVERY (PUBLIC) ==================
  if (path === "/api/channels/discover" && method === "GET") {
    try {
      const url = new URL(`https://dummy.com${path}`);
      const search = url.searchParams.get("search") || "";
      const category = url.searchParams.get("category") || "";
      const sort = url.searchParams.get("sort") || "popular";
      const limit = parseInt(url.searchParams.get("limit") || "20");
      const offset = parseInt(url.searchParams.get("offset") || "0");
      
      // Get all active channels
      let channels = await dbSelect("channels", { is_active: true });
      
      // Filter by search term
      if (search) {
        const searchLower = search.toLowerCase();
        channels = channels.filter((ch: Record<string, unknown>) => 
          (ch.channel_name as string).toLowerCase().includes(searchLower) ||
          (ch.description as string || '').toLowerCase().includes(searchLower)
        );
      }
      
      // Filter by category
      if (category) {
        channels = channels.filter((ch: Record<string, unknown>) => 
          ch.category === category
        );
      }
      
      // Enrich with stats
      const enriched = await Promise.all(channels.map(async (ch: Record<string, unknown>) => {
        try {
          const activeSubs = await dbSelect("subscriptions", { channel_id: ch.id as string, status: 'active' });
          const reviews = await dbSelect("reviews", { channel_id: ch.id as string });
          const visibleReviews = reviews.filter((r: Record<string, unknown>) => !r.is_hidden);
          const avgRating = visibleReviews.length > 0
            ? visibleReviews.reduce((sum: number, r: Record<string, unknown>) => sum + (r.rating as number), 0) / visibleReviews.length
            : 0;
          
          return {
            id: ch.id,
            channel_name: escapeHtml(ch.channel_name as string),
            description: escapeHtml(ch.description as string || ''),
            category: ch.category || 'other',
            subscription_price: ch.subscription_price,
            duration_days: ch.duration_days,
            is_verified: ch.is_verified,
            active_subscribers: activeSubs.length,
            avg_rating: Math.round(avgRating * 10) / 10,
            review_count: visibleReviews.length,
            created_at: ch.created_at,
          };
        } catch { return ch; }
      }));
      
      // Sort
      if (sort === "popular") {
        enriched.sort((a: any, b: any) => b.active_subscribers - a.active_subscribers);
      } else if (sort === "rating") {
        enriched.sort((a: any, b: any) => b.avg_rating - a.avg_rating);
      } else if (sort === "recent") {
        enriched.sort((a: any, b: any) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime());
      } else if (sort === "price_low") {
        enriched.sort((a: any, b: any) => parseFloat(a.subscription_price) - parseFloat(b.subscription_price));
      } else if (sort === "price_high") {
        enriched.sort((a: any, b: any) => parseFloat(b.subscription_price) - parseFloat(a.subscription_price));
      }
      
      // Paginate
      const paginated = enriched.slice(offset, offset + limit);
      
      return jsonResponse({
        channels: paginated,
        total: enriched.length,
        limit,
        offset
      }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  // ================== CATEGORIES ==================
  if (path === "/api/categories" && method === "GET") {
    try {
      const categories = [
        { id: "crypto", name: "Cryptocurrency", icon: "💰" },
        { id: "news", name: "News", icon: "📰" },
        { id: "education", name: "Education", icon: "📚" },
        { id: "entertainment", name: "Entertainment", icon: "🎬" },
        { id: "sports", name: "Sports", icon: "⚽" },
        { id: "technology", name: "Technology", icon: "💻" },
        { id: "business", name: "Business", icon: "💼" },
        { id: "lifestyle", name: "Lifestyle", icon: "🌟" },
        { id: "gaming", name: "Gaming", icon: "🎮" },
        { id: "music", name: "Music", icon: "🎵" }
      ];
      return jsonResponse(categories, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/channels/my" && method === "GET") {
    try {
      const channels = await dbSelect("channels", { owner_id: telegramId });
      const enriched = await Promise.all(channels.map(async (ch: Record<string, unknown>) => {
        try {
          const activeSubs = await dbSelect("subscriptions", { channel_id: ch.id as string, status: 'active' });
          const reviews = await dbSelect("reviews", { channel_id: ch.id as string });
          const visibleReviews = reviews.filter((r: Record<string, unknown>) => !r.is_hidden);
          const avgRating = visibleReviews.length > 0
            ? visibleReviews.reduce((sum: number, r: Record<string, unknown>) => sum + (r.rating as number), 0) / visibleReviews.length
            : 0;
          return {
            ...ch,
            channel_name: escapeHtml(ch.channel_name as string),
            active_subscribers: activeSubs.length,
            avg_rating: Math.round(avgRating * 10) / 10,
            review_count: visibleReviews.length,
          };
        } catch { return ch; }
      }));
      return jsonResponse(enriched, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/admin/channels" && method === "GET") {
    if (!isAdmin) return jsonResponse({ error: "Not found" }, 404, origin);
    try {
      const channels = await dbSelect("channels", {});
      const enriched = await Promise.all(channels.map(async (ch: Record<string, unknown>) => {
        try {
          const owners = await dbSelect("users", { telegram_id: ch.owner_id as number });
          const activeSubs = await dbSelect("subscriptions", { channel_id: ch.id as string, status: 'active' });
          const reviews = await dbSelect("reviews", { channel_id: ch.id as string });
          const visibleReviews = reviews.filter((r: Record<string, unknown>) => !r.is_hidden);
          const avgRating = visibleReviews.length > 0
            ? visibleReviews.reduce((sum: number, r: Record<string, unknown>) => sum + (r.rating as number), 0) / visibleReviews.length
            : 0;
          return {
            ...ch,
            owner_name: owners[0]?.first_name || 'Unknown',
            owner_username: owners[0]?.username || 'unknown',
            active_subscribers: activeSubs.length,
            avg_rating: Math.round(avgRating * 10) / 10,
            review_count: visibleReviews.length,
          };
        } catch { return ch; }
      }));
      return jsonResponse(enriched, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/channels/register" && method === "POST") {
    const { channel_name, channel_invite_link, subscription_price, duration_days } = body;
    if (!channel_name || !channel_invite_link) return jsonResponse({ error: "Name and invite link required" }, 400, origin);
    if (!subscription_price || (subscription_price as number) <= 0) return jsonResponse({ error: "Invalid price" }, 400, origin);
    const validDurations = [7, 30, 90, 180, 365, 9999];
    if (!duration_days || !validDurations.includes(duration_days as number)) {
      return jsonResponse({ error: "Invalid duration. Must be 7, 30, 90, 180, 365, or 9999 days" }, 400, origin);
    }
    try {
      const userChannels = await dbSelect("channels", { owner_id: telegramId });
      if (userChannels.length >= 5) {
        return jsonResponse({ error: "Maximum channel limit reached. You can only add up to 5 channels." }, 400, origin);
      }
      
      const botAdminCheck = await verifyBotIsAdmin(channel_invite_link as string);
      if (!botAdminCheck.success) {
        return jsonResponse({ 
          error: botAdminCheck.error || "Bot must be added as administrator to this channel first." 
        }, 400, origin);
      }
      
      const result = await dbInsert("channels", {
        owner_id: telegramId,
        channel_name,
        channel_invite_link,
        subscription_price,
        duration_days: duration_days,
        is_active: false,
      });
      return jsonResponse(result?.[0] || { success: true }, 201, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path.match(/^\/api\/channels\/[a-f0-9-]+$/) && method === "PUT") {
    const channelId = path.split("/")[3];
    if (!isValidUUID(channelId)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    const { subscription_price, duration_days } = body;
    try {
      const channels = await dbSelect("channels", { id: channelId, owner_id: telegramId });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found" }, 404, origin);
      const updateBody: Record<string, unknown> = { updated_at: new Date().toISOString() };
      if (subscription_price !== undefined && subscription_price !== null) {
        const parsedPrice = parseFloat(String(subscription_price));
        if (!isNaN(parsedPrice) && parsedPrice > 0) {
          updateBody.subscription_price = parsedPrice;
        }
      }
      if (duration_days !== undefined && duration_days !== null) {
        const parsedDuration = parseInt(String(duration_days));
        if (!isNaN(parsedDuration) && parsedDuration > 0) {
          updateBody.duration_days = parsedDuration;
        }
      }
      const result = await dbUpdate("channels", { id: channelId }, updateBody);
      return jsonResponse(result?.[0] || { success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path.match(/^\/api\/channels\/[a-f0-9-]+$/) && method === "GET") {
    const channelId = path.split("/")[3];
    if (!isValidUUID(channelId)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    try {
      const channels = await dbSelect("channels", { id: channelId, is_active: true });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found" }, 404, origin);
      
      const channel = channels[0];
      const owners = await dbSelect("users", { telegram_id: channel.owner_id as number });
      if (owners.length > 0 && owners[0].is_banned) {
        return jsonResponse({ error: "This channel is unavailable" }, 403, origin);
      }
      
      const reviews = await dbSelect("reviews", { channel_id: channelId });
      const visibleReviews = reviews.filter((r: Record<string, unknown>) => !r.is_hidden);
      
      return jsonResponse({ ...channel, reviews: visibleReviews }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path.match(/^\/api\/channels\/[a-f0-9-]+$/) && method === "DELETE") {
    const channelId = path.split("/")[3];
    if (!isValidUUID(channelId)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    try {
      const channels = await dbSelect("channels", { id: channelId, owner_id: telegramId });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found or unauthorized" }, 404, origin);
      
      await dbDelete("subscriptions", { channel_id: channelId });
      await dbDelete("reviews", { channel_id: channelId });
      await dbDelete("reports", { reported_channel_id: channelId });
      await dbDelete("platform_transactions", { channel_id: channelId });
      await dbDelete("channels", { id: channelId });
      
      console.log(`✅ Channel ${channelId} deleted by user ${telegramId}`);
      return jsonResponse({ success: true, message: "Channel deleted successfully" }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/subscriptions/my" && method === "GET") {
    try {
      const subs = await dbSelect("subscriptions", { user_id: telegramId });
      const enriched = await Promise.all(subs.map(async (sub: Record<string, unknown>) => {
        try {
          const channels = await dbSelect("channels", { id: sub.channel_id as string });
          const isExpired = sub.status === 'expired' || new Date(sub.end_date as string) < new Date();
          return { 
            ...sub, 
            channel: channels[0] || null,
            is_expired: isExpired
          };
        } catch { return { ...sub, channel: null, is_expired: false }; }
      }));
      return jsonResponse(enriched, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/subscriptions/initiate" && method === "POST") {
    const { channel_id, credits_used = 0 } = body;
    if (!channel_id || !isValidUUID(channel_id as string)) {
      return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    }

    const parsedCredits = Number(credits_used);
    if (!Number.isFinite(parsedCredits) || parsedCredits < 0) {
      return jsonResponse({ error: "Invalid credit amount" }, 400, origin);
    }
    const creditsRequested = Number(parsedCredits.toFixed(9));

    try {
      const channels = await dbSelect("channels", { id: channel_id as string, is_active: true });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found" }, 404, origin);
      const channel = channels[0];

      const ownerUsers = await dbSelect("users", { telegram_id: channel.owner_id as number });
      if (ownerUsers.length > 0 && ownerUsers[0].is_banned) {
        return jsonResponse({ error: "This channel is unavailable" }, 403, origin);
      }

      const existing = await dbSelect("subscriptions", { user_id: telegramId, channel_id: channel_id as string });
      if (existing.some((sub: Record<string, unknown>) =>
        sub.status === "active" && new Date(String(sub.end_date)) > new Date())) {
        return jsonResponse({ error: "Already subscribed" }, 400, origin);
      }

      // Allow one unexpired checkout per Telegram user. This prevents two tabs from
      // spending the same referral-credit balance against different payment intents.
      const activeIntents = await dbSelect("payment_nonces", {
        user_id: telegramId, used: false,
      });
      const nowMs = Date.now();
      const activeIntent = activeIntents.find((intent: Record<string, unknown>) =>
        Date.parse(String(intent.expires_at || "")) > nowMs);
      if (activeIntent) {
        const sameChannel = String(activeIntent.channel_id) === String(channel_id);
        return jsonResponse({
          error: sameChannel
            ? "A checkout is already pending for this channel. Finish it or wait for the payment reference to expire."
            : "You already have another checkout pending. Finish it or wait for its payment reference to expire.",
        }, 409, origin);
      }

      const channelPrice = Number(Number(channel.subscription_price).toFixed(9));
      const platformFee = Number((channelPrice * PLATFORM_FEE_PERCENT).toFixed(9));
      const totalPrice = Number((channelPrice + platformFee).toFixed(9));
      if (!Number.isFinite(channelPrice) || channelPrice <= 0 || !Number.isFinite(totalPrice)) {
        return jsonResponse({ error: "Invalid channel price" }, 400, origin);
      }
      if (creditsRequested > totalPrice + 1e-9) {
        return jsonResponse({ error: "Credits cannot exceed the subscription total" }, 400, origin);
      }

      const creditRows = creditsRequested > 0
        ? await dbSelect("referral_credits", { user_id: telegramId })
        : [];
      const availableCredits = creditRows.reduce(
        (sum: number, row: Record<string, unknown>) => sum + Math.max(0, Number(row.balance || 0)),
        0,
      );
      if (creditsRequested > availableCredits + 1e-9) {
        return jsonResponse({ error: "Insufficient referral credits" }, 400, origin);
      }
      if (!ADMIN_TON_WALLET) {
        return jsonResponse({ error: "Platform wallet is not configured" }, 500, origin);
      }

      // Network fees are paid separately by the sender's wallet. Do not add a fixed
      // 0.05 TON surcharge to the amount delivered to the escrow wallet.
      const amountRemaining = Number(Math.max(0, totalPrice - creditsRequested).toFixed(9));
      const amountNano = BigInt(Math.round(amountRemaining * 1e9));
      let recipientAddress: string;
      try {
        recipientAddress = Address.parse(ADMIN_TON_WALLET).toString({ urlSafe: true, bounceable: false });
      } catch {
        return jsonResponse({ error: "Platform wallet address is invalid" }, 500, origin);
      }

      const nonce = crypto.randomUUID();
      const expiresAt = new Date(Date.now() + 15 * 60 * 1000).toISOString();
      const payload = makeTonCommentPayload(nonce);

      await dbInsert("payment_nonces", {
        user_id: telegramId,
        channel_id: channel_id as string,
        nonce,
        expires_at: expiresAt,
        used: false,
        credits_used: creditsRequested,
        expected_total: totalPrice,
        channel_price: channelPrice,
        platform_fee: platformFee,
        duration_days: Number(channel.duration_days),
        expected_amount_nano: amountNano.toString(),
      });

      return jsonResponse({
        paymentDetails: {
          wallet: recipientAddress,
          amountNano: amountNano.toString(),
          amount: amountRemaining.toFixed(9),
          comment: nonce,
          payload,
          network: TON_NETWORK === "testnet" ? "-3" : "-239",
        },
        subscriptionDetails: {
          channel_id: channel.id,
          channel_name: channel.channel_name,
          channel_price: channelPrice,
          platform_fee: platformFee,
          total_price: totalPrice,
          credits_used: creditsRequested,
          remaining: amountRemaining,
          duration_days: channel.duration_days,
        },
        nonce,
        expires_at: expiresAt,
      }, 200, origin);
    } catch (e: unknown) {
      console.error("Payment initiation error:", e);
      return jsonResponse({ error: e instanceof Error ? e.message : "Failed" }, 500, origin);
    }
  }

  if (path === "/api/subscriptions/confirm" && method === "POST") {
    const { channel_id, nonce } = body;
    if (!channel_id || !isValidUUID(channel_id as string)) {
      return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    }
    if (typeof nonce !== "string" || nonce.length < 16 || nonce.length > 100) {
      return jsonResponse({ error: "Invalid payment reference" }, 400, origin);
    }

    try {
      const intents = await dbSelect("payment_nonces", {
        nonce,
        user_id: telegramId,
        channel_id: channel_id as string,
      });
      const intent = intents[0] as Record<string, unknown> | undefined;
      if (!intent) return jsonResponse({ error: "Payment intent not found" }, 404, origin);

      if (intent.used === true) {
        const alreadyCompleted = await dbSelect("subscriptions", {
          payment_nonce: nonce,
          user_id: telegramId,
        });
        if (alreadyCompleted.length > 0) {
          return jsonResponse({
            success: true,
            subscription: alreadyCompleted[0],
            end_date: alreadyCompleted[0].end_date,
            idempotent: true,
          }, 200, origin);
        }
        return jsonResponse({ error: "Payment reference has already been used" }, 409, origin);
      }

      const expiresAt = Date.parse(String(intent.expires_at || ""));
      if (!Number.isFinite(expiresAt) || expiresAt <= Date.now()) {
        return jsonResponse({ error: "Payment reference expired. Start checkout again." }, 410, origin);
      }

      const channels = await dbSelect("channels", { id: channel_id as string, is_active: true });
      if (channels.length === 0) return jsonResponse({ error: "Channel is unavailable" }, 404, origin);

      const expectedNano = BigInt(String(intent.expected_amount_nano || "0"));
      let transactionHash: string | null = null;
      if (expectedNano > 0n) {
        if (!ADMIN_TON_WALLET) return jsonResponse({ error: "Platform wallet is not configured" }, 500, origin);
        const createdAtMs = Date.parse(String(intent.created_at || "")) || (Date.now() - 15 * 60 * 1000);
        const verification = await verifyTonPayment(
          ADMIN_TON_WALLET,
          expectedNano,
          nonce,
          Math.floor(createdAtMs / 1000) - 30,
        );
        if (!verification.success || !verification.txHash) {
          return jsonResponse({
            error: verification.error || "Payment is not yet visible on-chain. Retry verification shortly.",
            pending: true,
          }, 402, origin);
        }
        transactionHash = verification.txHash;
      }

      const result = await dbRpc("complete_subscription_payment", {
        p_nonce: nonce,
        p_user_id: telegramId,
        p_transaction_hash: transactionHash,
      }) as Record<string, unknown>;

      const subscription = result?.subscription as Record<string, unknown> | undefined;
      if (!result?.success || !subscription) {
        throw new Error("Atomic payment completion did not return a subscription");
      }

      // Referral reward is best-effort; the payment/subscription/ledger transaction is already committed.
      try {
        const userRows = await dbSelect("users", { telegram_id: telegramId });
        const referrerId = userRows[0]?.referred_by;
        if (referrerId) {
          const channelPrice = Number(intent.channel_price || channels[0].subscription_price);
          await dbRpc("add_credits", {
            p_user_id: referrerId,
            p_amount: Number((channelPrice * REFERRAL_CREDIT_PERCENT).toFixed(9)),
            p_description: "Referral credit from subscription",
            p_reference_id: subscription.id,
          });
        }
      } catch (e) {
        console.error("Referral credit error:", e);
      }

      const channel = channels[0];
      if (channel.channel_chat_id) {
        try {
          await telegramApi("unbanChatMember", {
            chat_id: channel.channel_chat_id,
            user_id: telegramId,
            only_if_banned: true,
          });
        } catch (e) {
          console.error("Failed to restore subscriber access:", e);
        }
      }

      return jsonResponse({ success: true, subscription, end_date: subscription.end_date }, 200, origin);
    } catch (e: unknown) {
      console.error("Subscription confirmation error:", e);
      const message = e instanceof Error ? e.message : "Failed";
      if (/already used|already processed|duplicate|already subscribed/i.test(message)) {
        return jsonResponse({ error: message }, 409, origin);
      }
      if (/insufficient credits/i.test(message)) return jsonResponse({ error: message }, 400, origin);
      return jsonResponse({
        error: "Subscription confirmation failed. Check payment status before retrying.",
      }, 500, origin);
    }
  }

  if (path === "/api/cron/expire-subscriptions" && method === "POST") {
    if (!isAdmin) return jsonResponse({ error: "Unauthorized" }, 403, origin);
    
    try {
      const now = new Date().toISOString();
      const { data: expiredSubs, error } = await supabase
        .from('subscriptions')
        .select('id')
        .eq('status', 'active')
        .lt('end_date', now);
      
      if (error) throw error;
      if (!expiredSubs || expiredSubs.length === 0) {
        return jsonResponse({ success: true, expired: 0 }, 200, origin);
      }
      
      let expiredCount = 0;
      for (const sub of expiredSubs) {
        const result = await expireSubscription(sub.id);
        if (result.success) expiredCount++;
      }
      
      return jsonResponse({ success: true, expired: expiredCount }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/referrals/stats" && method === "GET") {
    try {
      if (!user.referral_code) {
        const code = generateReferralCode();
        await dbUpdate("users", { telegram_id: telegramId }, { referral_code: code });
        user.referral_code = code;
      }
      
      const referralCode = user.referral_code as string;
      const stats = await dbSelect("referrals", { referrer_id: telegramId });
      const credits = await dbSelect("referral_credits", { user_id: telegramId });
      const balance = credits[0]?.balance || 0;
      const totalEarned = credits[0]?.total_earned || 0;
      const totalUsed = credits[0]?.total_used || 0;
      const totalCreditsEarned = stats.reduce((sum: number, r: Record<string, unknown>) => sum + parseFloat(r.credits_earned as string || '0'), 0);
      
      return jsonResponse({
        referralCode,
        referralLink: `https://t.me/${BOT_TOKEN?.split(':')[0]}?start=ref_${referralCode}`,
        totalReferrals: stats.length,
        totalCreditsEarned,
        currentBalance: balance,
        totalUsed,
        referrals: stats,
      }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/withdrawals/my" && method === "GET") {
    try {
      const unpaid = await dbSelect("platform_transactions", {
        channel_owner_id: telegramId,
        owner_paid_out: false,
      });
      const grossRemaining = unpaid.reduce((sum: number, row: Record<string, unknown>) =>
        sum + Math.max(0, Number(row.owner_share || 0) - Number(row.paid_amount || 0)), 0);
      const withdrawals = await dbSelect("withdrawals", { channel_owner_id: telegramId });
      const activeWithdrawals = withdrawals.filter((row: Record<string, unknown>) =>
        ["pending", "approved"].includes(String(row.status)));
      const reserved = activeWithdrawals.reduce(
        (sum: number, row: Record<string, unknown>) => sum + Number(row.amount || 0), 0
      );

      return jsonResponse({
        // This project permits only one pending/approved withdrawal at a time.
        pendingEarnings: Number((activeWithdrawals.length ? 0 : grossRemaining).toFixed(9)),
        reservedWithdrawals: Number(reserved.toFixed(9)),
        withdrawals,
      }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : "Failed" }, 500, origin);
    }
  }

  if (path === "/api/withdrawals/request" && method === "POST") {
    const amount = Number(body.amount);
    const walletAddress = typeof body.wallet_address === "string" ? body.wallet_address.trim() : "";
    if (!Number.isFinite(amount) || amount < MIN_WITHDRAWAL) {
      return jsonResponse({ error: `Minimum withdrawal is ${MIN_WITHDRAWAL} TON` }, 400, origin);
    }
    if (!walletAddress || walletAddress.length > 128) {
      return jsonResponse({ error: "Valid wallet address required" }, 400, origin);
    }

    try {
      const canonicalWallet = Address.parse(walletAddress).toString({ urlSafe: true, bounceable: false });
      const withdrawal = await dbRpc("request_owner_withdrawal", {
        p_owner_id: telegramId,
        p_amount: Number(amount.toFixed(9)),
        p_wallet_address: canonicalWallet,
      });
      return jsonResponse({ success: true, withdrawal }, 200, origin);
    } catch (e: unknown) {
      const message = e instanceof Error ? e.message : "Failed";
      if (/insufficient balance|pending or approved withdrawal|already has a pending|minimum withdrawal/i.test(message)) {
        return jsonResponse({ error: message }, 400, origin);
      }
      console.error("Withdrawal request error:", e);
      return jsonResponse({ error: "Unable to reserve this withdrawal. Check your balance and wallet address." }, 500, origin);
    }
  }

  if (path === "/api/reviews" && method === "POST") {
    const { channel_id, rating, comment } = body;
    if (!channel_id || !rating || (rating as number) < 1 || (rating as number) > 5) return jsonResponse({ error: "Invalid rating" }, 400, origin);
    try {
      const existing = await dbSelect("reviews", { user_id: telegramId, channel_id: channel_id as string });
      if (existing.length > 0) return jsonResponse({ error: "Already rated" }, 400, origin);
      await dbInsert("reviews", { user_id: telegramId, channel_id, rating, comment: comment || null });
      return jsonResponse({ success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path.match(/^\/api\/reviews\/[a-f0-9-]+$/) && method === "GET") {
    const channelId = path.split("/").pop()!;
    if (!isValidUUID(channelId)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    try {
      const reviews = await dbSelect("reviews", { channel_id: channelId });
      return jsonResponse(reviews.filter((r: Record<string, unknown>) => !r.is_hidden), 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/reports" && method === "POST") {
    const { channel_id, reason, description } = body;
    if (!channel_id || !reason) return jsonResponse({ error: "Channel and reason required" }, 400, origin);
    try {
      const existing = await dbSelect("reports", { reporter_id: telegramId, reported_channel_id: channel_id as string });
      if (existing.length > 0) return jsonResponse({ error: "Already reported" }, 400, origin);
      await dbInsert("reports", { reporter_id: telegramId, reported_channel_id: channel_id, reason, description: description || null });
      return jsonResponse({ success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (!isAdmin) return jsonResponse({ error: "Not found" }, 404, origin);
  
  if (path === "/api/admin/reports" && method === "GET") {
    try {
      const reports = await dbSelect("reports", {});
      const enriched = await Promise.all(reports.map(async (report: Record<string, unknown>) => {
        try {
          const reporters = await dbSelect("users", { telegram_id: report.reporter_id as number });
          const channels = await dbSelect("channels", { id: report.reported_channel_id as string });
          let channelOwner = { first_name: 'Unknown' };
          if (channels[0]?.owner_id) {
            const owners = await dbSelect("users", { telegram_id: channels[0].owner_id as number });
            channelOwner = owners[0] || channelOwner;
          }
          return { ...report, reporter: reporters[0] || { first_name: 'Unknown' }, channel: channels[0] || { channel_name: 'Unknown' }, channel_owner: channelOwner };
        } catch { return report; }
      }));
      return jsonResponse(enriched, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path.match(/^\/api\/admin\/reports\/[a-f0-9-]+\/review$/) && method === "POST") {
    const reportId = path.split("/")[4];
    const { action } = body;
    if (!isValidUUID(reportId)) return jsonResponse({ error: "Invalid report ID" }, 400, origin);
    if (!action || !['ban', 'dismiss'].includes(action as string)) return jsonResponse({ error: "Invalid action" }, 400, origin);
    try {
      const reports = await dbSelect("reports", { id: reportId });
      const report = reports[0];
      if (!report) return jsonResponse({ error: "Report not found" }, 404, origin);
      if (action === 'dismiss') {
        await dbDelete("reports", { id: reportId });
        return jsonResponse({ success: true, action: 'dismissed' }, 200, origin);
      }
      if (action === 'ban') {
        await dbUpdate("reports", { id: reportId }, { status: 'banned', reviewed_by: telegramId, reviewed_at: new Date().toISOString() });
        await dbUpdate("channels", { id: report.reported_channel_id as string }, { is_active: false });
        const channels = await dbSelect("channels", { id: report.reported_channel_id as string });
        if (channels.length > 0) {
          await dbUpdate("users", { telegram_id: channels[0].owner_id as number }, { is_banned: true, banned_at: new Date().toISOString() });
        }
        return jsonResponse({ success: true, action: 'banned' }, 200, origin);
      }
      return jsonResponse({ error: "Invalid action" }, 400, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }
  
  if (path === "/api/admin/withdrawals" && method === "GET") {
    try {
      const pending = await dbSelect("withdrawals", { status: "pending" });
      const approved = await dbSelect("withdrawals", { status: "approved" });
      return jsonResponse([...pending, ...approved], 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : "Failed" }, 500, origin);
    }
  }

  // Approval reserves funds but does not mark a payout as completed.
  if (path.match(/^\/api\/admin\/withdrawals\/[a-f0-9-]+\/approve$/) && method === "POST") {
    const withdrawalId = path.split("/")[4];
    if (!isValidUUID(withdrawalId)) return jsonResponse({ error: "Invalid withdrawal ID" }, 400, origin);
    try {
      const withdrawals = await dbSelect("withdrawals", { id: withdrawalId });
      const withdrawal = withdrawals[0] as Record<string, unknown> | undefined;
      if (!withdrawal || withdrawal.status !== "pending") {
        return jsonResponse({ error: "Withdrawal is not pending" }, 409, origin);
      }
      const wallet = String(withdrawal.wallet_address || "");
      try { Address.parse(wallet); } catch { return jsonResponse({ error: "Invalid withdrawal wallet address" }, 400, origin); }

      const updated = await dbUpdate("withdrawals", { id: withdrawalId, status: "pending" }, {
        status: "approved",
        approved_at: new Date().toISOString(),
        approved_by: telegramId,
      });
      if (!updated.length) return jsonResponse({ error: "Withdrawal was already changed" }, 409, origin);

      return jsonResponse({
        success: true,
        status: "approved",
        transfer_instructions: {
          from: ADMIN_TON_WALLET,
          to: wallet,
          amount: Number(withdrawal.amount),
          amount_nano: BigInt(Math.round(Number(withdrawal.amount) * 1e9)).toString(),
          note: "Send this exact amount from the platform wallet, then submit the outgoing transaction hash to the complete endpoint. Approval alone does not mark funds as paid.",
        },
      }, 200, origin);
    } catch (e: unknown) {
      console.error("Withdrawal approval error:", e);
      return jsonResponse({ error: "Unable to approve withdrawal" }, 500, origin);
    }
  }

  if (path.match(/^\/api\/admin\/withdrawals\/[a-f0-9-]+\/complete$/) && method === "POST") {
    const withdrawalId = path.split("/")[4];
    const transactionHash = typeof body.transaction_hash === "string" ? body.transaction_hash.trim() : "";
    if (!isValidUUID(withdrawalId)) return jsonResponse({ error: "Invalid withdrawal ID" }, 400, origin);
    if (!transactionHash || transactionHash.length > 200) {
      return jsonResponse({ error: "Outgoing transaction hash is required" }, 400, origin);
    }

    try {
      const withdrawals = await dbSelect("withdrawals", { id: withdrawalId });
      const withdrawal = withdrawals[0] as Record<string, unknown> | undefined;
      if (!withdrawal) return jsonResponse({ error: "Withdrawal not found" }, 404, origin);
      if (withdrawal.status === "processed" && withdrawal.payout_tx_hash === transactionHash) {
        return jsonResponse({ success: true, withdrawal, idempotent: true }, 200, origin);
      }
      if (withdrawal.status !== "approved") {
        return jsonResponse({ error: "Withdrawal must be approved before completion" }, 409, origin);
      }

      const wallet = String(withdrawal.wallet_address || "");
      const amountNano = BigInt(Math.round(Number(withdrawal.amount) * 1e9));
      const approvedAtMs = Date.parse(String(withdrawal.approved_at || withdrawal.request_date || "")) || 0;
      const isVerified = await verifyTonPayout(
        transactionHash,
        wallet,
        amountNano,
        Math.floor(approvedAtMs / 1000) - 30,
      );
      if (!isVerified) {
        return jsonResponse({ error: "Outgoing transaction not found or amount/recipient does not match" }, 400, origin);
      }

      const result = await dbRpc("complete_withdrawal", {
        p_withdrawal_id: withdrawalId,
        p_admin_id: telegramId,
        p_transaction_hash: transactionHash,
      });
      return jsonResponse({ success: true, withdrawal: result }, 200, origin);
    } catch (e: unknown) {
      console.error("Withdrawal completion error:", e);
      return jsonResponse({ error: "Unable to complete withdrawal. Verify the outgoing transaction and retry." }, 500, origin);
    }
  }

  return jsonResponse({ error: "Not found" }, 404, origin);
}

// ================== MAIN SERVER ==================
serve(async (req: Request) => {
  const url = new URL(req.url);
  const path = url.pathname;
  const method = req.method;
  const origin = req.headers.get("Origin");
  
  if (method === "OPTIONS") return new Response(null, { headers: getCorsHeaders(origin) });
  
  if (path === "/health" || path === "/" || path.endsWith("/health") || path.endsWith("/mainbot")) {
    try {
      const { error } = await supabase.from('users').select('telegram_id').limit(1);
      if (error) throw error;
      
      let botStatus = 'unknown';
      try {
        const botInfo = await telegramApi("getMe");
        botStatus = botInfo.ok ? `✅ @${botInfo.result.username}` : `❌ ${botInfo.description}`;
      } catch (e) {
        botStatus = `❌ ${e instanceof Error ? e.message : 'Unknown error'}`;
      }
      
      return jsonResponse({ 
        status: 'healthy', 
        version: '2.5.1', 
        timestamp: new Date().toISOString(), 
        network: TON_NETWORK,
        bot: botStatus,
        webhook_secret_set: !!WEBHOOK_SECRET
      }, 200, origin);
    } catch {
      return jsonResponse({ status: 'unhealthy', error: 'Database connection failed' }, 503, origin);
    }
  }
  
  if (path.endsWith("/api/cron/expire-subscriptions") && method === "POST") {
    const suppliedSecret = req.headers.get("x-cron-secret") || "";
    if (!CRON_SECRET || suppliedSecret !== CRON_SECRET) {
      return jsonResponse({ error: "Unauthorized" }, 403, origin);
    }
    try {
      const now = new Date().toISOString();
      const { data: expiredSubs, error } = await supabase
        .from("subscriptions")
        .select("id")
        .eq("status", "active")
        .lt("end_date", now);
      if (error) throw error;
      let expiredCount = 0;
      for (const sub of expiredSubs || []) {
        const result = await expireSubscription(sub.id);
        if (result.success) expiredCount++;
      }
      return jsonResponse({ success: true, expired: expiredCount }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : "Failed" }, 500, origin);
    }
  }

  if (path.endsWith("/test-bot") && method === "GET") {
    const diagnosticUser = await validateInitData(req.headers.get("x-telegram-initdata") || "");
    if (!diagnosticUser?.id || String(diagnosticUser.id) !== String(ADMIN_TELEGRAM_ID)) {
      return jsonResponse({ error: "Not found" }, 404, origin);
    }
    const chatId = url.searchParams.get("chat_id");
    const message = url.searchParams.get("message") || "Test message";
    if (!chatId || !/^-?\d+$/.test(chatId)) return jsonResponse({ error: "Invalid chat_id" }, 400, origin);
    try {
      const result = await telegramApi("sendMessage", { chat_id: parseInt(chatId), text: message });
      return jsonResponse({ success: result.ok, message_id: result.result?.message_id, error: result.ok ? null : result.description }, result.ok ? 200 : 500, origin);
    } catch (error) {
      return jsonResponse({ error: "Failed", details: error instanceof Error ? error.message : "Unknown" }, 500, origin);
    }
  }
  
  if (path.endsWith("/bot-webhook") && method === "POST") {
    console.log("📩 Webhook received");
    if (!verifyWebhookSecret(req.headers)) {
      console.error("❌ Webhook secret verification failed");
      return new Response("Unauthorized", { status: 401, headers: getCorsHeaders(origin) });
    }
    try {
      const body = await req.json();
      console.log("📩 Webhook update type:", Object.keys(body).filter(k => body[k] !== undefined).join(", "));
      await handleBotUpdate(body);
      return new Response("OK", { headers: getCorsHeaders(origin) });
    } catch (e) {
      console.error("❌ Bot webhook error:", e);
      return new Response("Error", { status: 500, headers: getCorsHeaders(origin) });
    }
  }
  
  if (path.includes("/api/")) {
    try {
      const apiPath = path.substring(path.indexOf("/api/"));
      let body: Record<string, unknown> = {};
      if (["POST", "PUT", "PATCH"].includes(method)) {
        try { body = await req.json(); } catch { return jsonResponse({ error: "Invalid JSON" }, 400, origin); }
      }
      return await apiHandler(apiPath, method, req.headers, body, origin);
    } catch (e) {
      console.error("API error:", e);
      return jsonResponse({ error: "Internal server error" }, 500, origin);
    }
  }
  
  return new Response("MySubHub API v2.5.1 - ESCROW HARDENED", { headers: getCorsHeaders(origin) });
});
