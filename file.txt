// ================== DB HELPERS ==================
async function dbSelect(table: string, filters: Record<string, unknown> = {}, select = '*') {
  let query = supabase.from(table).select(select);
  for (const [key, value] of Object.entries(filters)) {
    if (value !== undefined && value !== null) {
      query = query.eq(key, value);
    }
  }
  const { data, error } = await query;
  if (error) throw new Error(`DB select error [${table}]: ${error.message}`);
  return data || [];
}

async function dbInsert(table: string, body: Record<string, unknown>) {
  const { data, error } = await supabase.from(table).insert(body).select();
  if (error) throw new Error(`DB insert error [${table}]: ${error.message}`);
  return data;
}

async function dbUpdate(table: string, filters: Record<string, unknown>, body: Record<string, unknown>) {
  let query = supabase.from(table).update(body);
  for (const [key, value] of Object.entries(filters)) {
    if (value !== undefined && value !== null) {
      query = query.eq(key, value);
    }
  }
  const { data, error } = await query.select();
  if (error) throw new Error(`DB update error [${table}]: ${error.message}`);
  return data;
}

async function dbDelete(table: string, filters: Record<string, unknown>) {
  let query = supabase.from(table).delete();
  for (const [key, value] of Object.entries(filters)) {
    if (value !== undefined && value !== null) {
      query = query.eq(key, value);
    }
  }
  const { error } = await query;
  if (error) throw new Error(`DB delete error [${table}]: ${error.message}`);
  return true;
}

async function dbRpc(fn: string, params: Record<string, unknown> = {}) {
  const { data, error } = await supabase.rpc(fn, params);
  if (error) throw new Error(`RPC error [${fn}]: ${error.message}`);
  return data;
}

// ================== UTILITIES ==================
function isValidUUID(str: string): boolean {
  return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(str);
}

function escapeHtml(str: string): string {
  if (!str) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#x27;');
}

function generateReferralCode(): string {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  let code = '';
  for (let i = 0; i < 8; i++) code += chars.charAt(Math.floor(Math.random() * chars.length));
  return code;
}

// ================== CORS ==================
const ALLOWED_ORIGINS = ["https://haman-hub.github.io", "https://web.telegram.org", "https://t.me"];

function getCorsHeaders(origin: string | null) {
  const allowed = origin && ALLOWED_ORIGINS.some(o => origin.startsWith(o)) ? origin : ALLOWED_ORIGINS[0];
  return {
    "Access-Control-Allow-Origin": allowed,
    "Access-Control-Allow-Methods": "GET, POST, PUT, PATCH, DELETE, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type, x-telegram-initdata, Authorization",
    "Access-Control-Max-Age": "86400",
    "Vary": "Origin",
  };
}

function jsonResponse(payload: unknown, status = 200, origin: string | null = null) {
  return new Response(JSON.stringify(payload), {
    status,
    headers: { "Content-Type": "application/json", ...getCorsHeaders(origin) },
  });
}

// ================== RATE LIMITING ==================
const rateLimitMap = new Map<string, { count: number; resetAt: number }>();
setInterval(() => {
  const now = Date.now();
  for (const [key, entry] of rateLimitMap) {
    if (now > entry.resetAt) rateLimitMap.delete(key);
  }
}, 300_000);

function checkRateLimit(userId: string): boolean {
  const now = Date.now();
  const entry = rateLimitMap.get(userId);
  if (!entry || now > entry.resetAt) {
    rateLimitMap.set(userId, { count: 1, resetAt: now + 60000 });
    return true;
  }
  if (entry.count >= 30) return false;
  entry.count++;
  return true;
}

// ================== TELEGRAM ==================
async function telegramApi(method: string, body: Record<string, unknown> = {}) {
  if (!BOT_TOKEN) throw new Error("BOT_TOKEN not configured");
  const res = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/${method}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  return res.json();
}

// ================== INITDATA VALIDATION ==================
async function validateInitData(initData: string): Promise<Record<string, unknown> | null> {
  try {
    if (!initData || !BOT_TOKEN) return null;
    const params = new URLSearchParams(initData);
    const authDate = parseInt(params.get("auth_date") || "0", 10);
    const now = Math.floor(Date.now() / 1000);
    if (!authDate || now - authDate > 86400) return null;

    const secretKey = await crypto.subtle.importKey(
      "raw", new TextEncoder().encode("WebAppData"),
      { name: "HMAC", hash: "SHA-256" }, false, ["sign"]
    );
    const key = await crypto.subtle.sign("HMAC", secretKey, new TextEncoder().encode(BOT_TOKEN));
    const hash = params.get("hash");
    params.delete("hash");
    const sorted = new URLSearchParams();
    Array.from(params.keys()).sort().forEach(k => sorted.set(k, params.get(k)!));
    const dataCheckString = Array.from(sorted.entries()).map(([k, v]) => `${k}=${v}`).join("\n");
    const hmac = await crypto.subtle.sign(
      "HMAC",
      await crypto.subtle.importKey("raw", key, { name: "HMAC", hash: "SHA-256" }, false, ["sign"]),
      new TextEncoder().encode(dataCheckString)
    );
    const calculated = Array.from(new Uint8Array(hmac)).map(b => b.toString(16).padStart(2, "0")).join("");
    if (calculated !== hash) return null;

    const userJson = params.get("user");
    if (!userJson) return null;
    return JSON.parse(userJson);
  } catch (e) {
    console.error("InitData validation error:", e);
    return null;
  }
}

// ================== WEBHOOK VERIFICATION ==================
function verifyWebhookSecret(headers: Headers): boolean {
  if (!WEBHOOK_SECRET) {
    console.error("🚨 WEBHOOK_SECRET not configured! Rejecting webhook.");
    return false;
  }
  return headers.get("X-Telegram-Bot-Api-Secret-Token") === WEBHOOK_SECRET;
}

// ================== TON PAYMENT VERIFICATION ==================
async function verifyTonPayment(walletAddress: string, expectedAmountNano: bigint, comment?: string) {
  try {
    const txRes = await fetch(`${TONCENTER_BASE}/getTransactions?address=${walletAddress}&limit=50`, {
      headers: { "X-API-Key": TONCENTER_API_KEY || "" }
    });
    const txData = await txRes.json();
    if (!txData.ok || !txData.result) return { success: false, error: "Failed to fetch transactions" };

    const cutoff = Math.floor(Date.now() / 1000) - 600;
    for (const tx of txData.result) {
      if (tx.utime < cutoff) continue;
      const inMsg = tx.in_msg;
      if (!inMsg?.value) continue;
      try {
        const received = BigInt(inMsg.value);
        if (received >= expectedAmountNano) {
          if (comment && inMsg.message !== comment) continue;
          const txHash = tx.transaction_id?.hash;
          if (!txHash) continue;
          const existing = await dbSelect("subscriptions", { transaction_hash: txHash });
          if (existing.length > 0) continue;
          return { success: true, txHash };
        }
      } catch { continue; }
    }
    return { success: false, error: "No matching transaction found" };
  } catch (e: unknown) {
    return { success: false, error: `Verification error: ${e instanceof Error ? e.message : 'Unknown'}` };
  }
}

// ================== BOT WEBHOOK HANDLER ==================
async function handleBotUpdate(body: Record<string, unknown>) {
  const message = body.message as Record<string, unknown> | undefined;
  if (message?.text && (message.text as string).startsWith("/start")) {
    const chatId = (message.chat as Record<string, unknown>).id as number;
    const from = message.from as Record<string, unknown>;
    const telegramId = from.id as number;

    try {
      const existing = await dbSelect("users", { telegram_id: telegramId });
      if (existing.length === 0) {
        await dbInsert("users", {
          telegram_id: telegramId,
          first_name: from.first_name,
          last_name: from.last_name,
          username: from.username,
          language_code: from.language_code || 'en',
          referral_code: generateReferralCode(),
        });
        await dbInsert("user_preferences", { user_id: telegramId, theme: 'dark' }).catch(() => {});
      } else {
        await dbUpdate("users", { telegram_id: telegramId }, {
          first_name: from.first_name,
          last_name: from.last_name,
          username: from.username,
        });
      }
    } catch (e) {
      console.error("User upsert error:", e);
    }

    const text = message.text as string;
    const parts = text.replace(/^\/start(?:@\w+)?\s*/, "").trim();
    let url = WEBAPP_URL;
    let msgText = `Welcome, ${escapeHtml(from.first_name as string || 'User')}! 🎉`;
    let btnText = "🚀 Open Dashboard";

    if (parts === "owner") { url += "?startapp=owner"; btnText = "📺 My Channels"; }
    else if (parts.startsWith("ref_")) { url += "?startapp=referral"; }
    else if (parts && isValidUUID(parts)) { url += `?startapp=${parts}`; msgText = "Subscribe to this channel:"; btnText = "💎 Subscribe"; }

    await telegramApi("sendMessage", {
      chat_id: chatId,
      text: msgText,
      reply_markup: { inline_keyboard: [[{ text: btnText, web_app: { url } }]] }
    });
  }

  // Handle bot added to channels/groups
  const myChatMember = body.my_chat_member as Record<string, unknown> | undefined;
  if (myChatMember) {
    const chat = myChatMember.chat as Record<string, unknown>;
    const newStatus = (myChatMember.new_chat_member as Record<string, unknown>)?.status;
    const from = myChatMember.from as Record<string, unknown>;

    if (newStatus === "administrator" && ["channel", "group", "supergroup"].includes(chat.type as string)) {
      let inviteLink = "";
      try {
        const res = await telegramApi("exportChatInviteLink", { chat_id: chat.id });
        if (res.ok) inviteLink = res.result as string;
      } catch {}

      try {
        const existing = await dbSelect("channels", { channel_chat_id: chat.id as number });
        if (existing.length === 0) {
          await dbInsert("channels", {
            owner_id: from.id,
            channel_name: chat.title || "Unknown",
            channel_type: chat.type === "channel" ? "channel" : "group",
            channel_chat_id: chat.id,
            channel_invite_link: inviteLink,
            subscription_price: 0.1,
            duration_days: 30,
            is_active: false,
          });
        } else {
          await dbUpdate("channels", { channel_chat_id: chat.id as number }, {
            channel_chat_id: chat.id,
            channel_invite_link: inviteLink || undefined,
            is_active: true,
          });
        }
      } catch (e) {
        console.error("Channel registration error:", e);
      }
    }
  }
}

// ================== API HANDLER ==================
async function apiHandler(path: string, method: string, headers: Headers, body: Record<string, unknown>, origin: string | null) {
  // Authenticate
  const initData = headers.get("x-telegram-initdata") || "";
  const tgUser = await validateInitData(initData);

  if (!tgUser || !tgUser.id) {
    return jsonResponse({ error: "Unauthorized: Invalid Telegram data" }, 401, origin);
  }

  // Use telegram_id as the user identifier (matches database schema)
  const telegramId = tgUser.id as number;

  // Get or create user
  let user: Record<string, unknown>;
  try {
    const users = await dbSelect("users", { telegram_id: telegramId });
    if (users.length === 0) {
      const referralCode = generateReferralCode();
      await dbInsert("users", {
        telegram_id: telegramId,
        first_name: tgUser.first_name,
        last_name: tgUser.last_name,
        username: tgUser.username,
        language_code: (tgUser.language_code as string) || 'en',
        referral_code: referralCode,
      });
      user = { telegram_id: telegramId, first_name: tgUser.first_name, last_name: tgUser.last_name, username: tgUser.username, referral_code: referralCode, is_banned: false, role: 'user' };
    } else {
      user = users[0];
    }
  } catch (e) {
    console.error("User auth error:", e);
    return jsonResponse({ error: "Authentication failed" }, 500, origin);
  }

  // Check banned
  if (user.is_banned) {
    return jsonResponse({ error: "Account suspended" }, 403, origin);
  }

  // Rate limit using telegram_id
  const userIdStr = telegramId.toString();
  if (!checkRateLimit(userIdStr)) {
    return jsonResponse({ error: "Too many requests" }, 429, origin);
  }

  // Admin check using telegram_id
  const isAdmin = userIdStr === ADMIN_TELEGRAM_ID;

  // ================== AUTH ==================
  if (path === "/api/auth/validate" && method === "GET") {
    return jsonResponse({
      user: {
        telegram_id: user.telegram_id,
        first_name: user.first_name,
        last_name: user.last_name,
        username: user.username,
        wallet_address: user.wallet_address,
        role: user.role,
        referral_code: user.referral_code,
      },
      isAdmin,
    }, 200, origin);
  }

  if (path === "/api/auth/wallet" && method === "POST") {
    const { wallet_address } = body;
    if (!wallet_address || typeof wallet_address !== 'string') {
      return jsonResponse({ error: "Invalid wallet address" }, 400, origin);
    }
    await dbUpdate("users", { telegram_id: telegramId }, { wallet_address });
    return jsonResponse({ success: true }, 200, origin);
  }

  // ================== CHANNELS ==================
  if (path === "/api/channels/my" && method === "GET") {
    try {
      const channels = await dbSelect("channels", { owner_id: telegramId });
      const enriched = await Promise.all(channels.map(async (ch: Record<string, unknown>) => {
        try {
          const activeSubs = await dbSelect("subscriptions", { channel_id: ch.id as string, status: 'active' });
          const reviews = await dbSelect("reviews", { channel_id: ch.id as string });
          const visibleReviews = reviews.filter((r: Record<string, unknown>) => !r.is_hidden);
          const avgRating = visibleReviews.length > 0
            ? visibleReviews.reduce((sum: number, r: Record<string, unknown>) => sum + (r.rating as number), 0) / visibleReviews.length
            : 0;
          return {
            ...ch,
            channel_name: escapeHtml(ch.channel_name as string),
            active_subscribers: activeSubs.length,
            avg_rating: Math.round(avgRating * 10) / 10,
            review_count: visibleReviews.length,
          };
        } catch { return ch; }
      }));
      return jsonResponse(enriched, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed to load channels' }, 500, origin);
    }
  }

  if (path === "/api/channels/register" && method === "POST") {
    const { channel_name, channel_invite_link, subscription_price, duration_days } = body;
    if (!channel_name || !channel_invite_link) return jsonResponse({ error: "Name and invite link required" }, 400, origin);
    if (!subscription_price || (subscription_price as number) <= 0) return jsonResponse({ error: "Invalid price" }, 400, origin);
    try {
      const result = await dbInsert("channels", {
        owner_id: telegramId,
        channel_name,
        channel_invite_link,
        subscription_price,
        duration_days: duration_days || 30,
        is_active: false,
      });
      return jsonResponse(result?.[0] || { success: true }, 201, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed to register channel' }, 500, origin);
    }
  }

  if (path.match(/^\/api\/channels\/[a-f0-9-]+$/) && method === "PUT") {
    const channelId = path.split("/")[3];
    if (!isValidUUID(channelId)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    const { subscription_price, duration_days } = body;
    try {
      const channels = await dbSelect("channels", { id: channelId, owner_id: telegramId });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found" }, 404, origin);
      const updateBody: Record<string, unknown> = { updated_at: new Date().toISOString() };
      if (subscription_price && (subscription_price as number) > 0) updateBody.subscription_price = subscription_price;
      if (duration_days && (duration_days as number) > 0) updateBody.duration_days = duration_days;
      const result = await dbUpdate("channels", { id: channelId }, updateBody);
      return jsonResponse(result?.[0] || { success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed to update channel' }, 500, origin);
    }
  }

  if (path.match(/^\/api\/channels\/[a-f0-9-]+$/) && method === "GET") {
    const channelId = path.split("/")[3];
    if (!isValidUUID(channelId)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    try {
      const channels = await dbSelect("channels", { id: channelId, is_active: true });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found" }, 404, origin);
      return jsonResponse(channels[0], 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  // ================== SUBSCRIPTIONS ==================
  if (path === "/api/subscriptions/my" && method === "GET") {
    try {
      const subs = await dbSelect("subscriptions", { user_id: telegramId });
      const enriched = await Promise.all(subs.map(async (sub: Record<string, unknown>) => {
        try {
          const channels = await dbSelect("channels", { id: sub.channel_id as string });
          return { ...sub, channel: channels[0] || null };
        } catch { return { ...sub, channel: null }; }
      }));
      return jsonResponse(enriched, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed to load subscriptions' }, 500, origin);
    }
  }

  if (path === "/api/subscriptions/initiate" && method === "POST") {
    const { channel_id, credits_used = 0 } = body;
    if (!channel_id || !isValidUUID(channel_id as string)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    try {
      const channels = await dbSelect("channels", { id: channel_id as string, is_active: true });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found" }, 404, origin);
      const channel = channels[0];

      const existing = await dbSelect("subscriptions", { user_id: telegramId, channel_id: channel_id as string, status: 'active' });
      if (existing.length > 0) {
        const endDate = new Date(existing[0].end_date as string);
        if (endDate > new Date()) return jsonResponse({ error: "Already subscribed" }, 400, origin);
      }

      const totalPrice = parseFloat(channel.subscription_price as string);
      const creditsAmount = parseFloat(credits_used as string) || 0;
      const remaining = totalPrice - creditsAmount;
      const remainingNano = BigInt(Math.round(remaining * 1e9));
      const totalNano = remainingNano + NETWORK_FEE_NANO;

      const owners = await dbSelect("users", { telegram_id: channel.owner_id as number });
      const ownerWallet = owners[0]?.wallet_address;
      if (!ownerWallet) return jsonResponse({ error: "Owner wallet not configured" }, 400, origin);

      const nonce = crypto.randomUUID();
      await dbInsert("payment_nonces", {
        user_id: telegramId,
        channel_id: channel_id as string,
        nonce,
        expires_at: new Date(Date.now() + 15 * 60 * 1000).toISOString(),
      });

      return jsonResponse({
        paymentDetails: { wallet: ownerWallet, amountNano: totalNano.toString(), amount: remaining.toFixed(9), comment: nonce },
        subscriptionDetails: { channel_id: channel.id, channel_name: channel.channel_name, price: totalPrice, credits_used: creditsAmount, remaining, duration_days: channel.duration_days },
        nonce,
      }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path === "/api/subscriptions/confirm" && method === "POST") {
    const { channel_id, transaction_hash, credits_used = 0, nonce } = body;
    if (!channel_id || !isValidUUID(channel_id as string)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    try {
      const channels = await dbSelect("channels", { id: channel_id as string, is_active: true });
      if (channels.length === 0) return jsonResponse({ error: "Channel not found" }, 404, origin);
      const channel = channels[0];

      const totalPrice = parseFloat(channel.subscription_price as string);
      const creditsAmount = parseFloat(credits_used as string) || 0;
      const remaining = totalPrice - creditsAmount;

      // Deduct credits atomically
      if (creditsAmount > 0) {
        try {
          const success = await dbRpc('deduct_credits', { p_user_id: telegramId, p_amount: creditsAmount });
          if (!success) return jsonResponse({ error: "Insufficient credits" }, 400, origin);
        } catch (e) {
          return jsonResponse({ error: "Credit deduction failed" }, 500, origin);
        }
      }

      // Verify payment
      let txHash = transaction_hash as string | undefined;
      if (remaining > 0 && !txHash) {
        const owners = await dbSelect("users", { telegram_id: channel.owner_id as number });
        const ownerWallet = owners[0]?.wallet_address;
        if (ownerWallet) {
          const result = await verifyTonPayment(ownerWallet, BigInt(Math.round(remaining * 1e9)), nonce as string);
          if (!result.success) return jsonResponse({ error: result.error }, 400, origin);
          txHash = result.txHash;
        }
      }
      if (!txHash) return jsonResponse({ error: "No valid transaction" }, 400, origin);

      // Check duplicate
      const dup = await dbSelect("subscriptions", { transaction_hash: txHash });
      if (dup.length > 0) return jsonResponse({ error: "Transaction already processed" }, 409, origin);

      // Create subscription
      const endDate = new Date();
      endDate.setDate(endDate.getDate() + (channel.duration_days as number));
      const subResult = await dbInsert("subscriptions", {
        user_id: telegramId,
        channel_id: channel_id as string,
        end_date: endDate.toISOString(),
        transaction_hash: txHash,
        amount: totalPrice,
        status: 'active',
        credits_used: creditsAmount,
      });

      // Record platform transaction
      const platformFee = totalPrice * PLATFORM_FEE_PERCENT;
      const ownerShare = totalPrice - platformFee;
      await dbInsert("platform_transactions", {
        subscription_id: subResult?.[0]?.id,
        user_id: telegramId,
        channel_id: channel_id as string,
        channel_owner_id: channel.owner_id,
        amount_total: totalPrice,
        owner_share: ownerShare,
        platform_fee: platformFee,
        network_fee: Number(NETWORK_FEE_NANO) / 1e9,
        transaction_hash: txHash,
      });

      // Mark nonce used
      if (nonce) {
        await dbUpdate("payment_nonces", { nonce: nonce as string }, { used: true }).catch(() => {});
      }

      // Referral credits
      try {
        const referrers = await dbSelect("users", { telegram_id: telegramId });
        const referrerId = referrers[0]?.referred_by;
        if (referrerId) {
          const credit = totalPrice * REFERRAL_CREDIT_PERCENT;
          await dbRpc('add_credits', { p_user_id: referrerId, p_amount: credit, p_description: 'Referral credit' }).catch(() => {});
        }
      } catch {}

      return jsonResponse({ success: true, subscription: subResult?.[0], end_date: endDate.toISOString() }, 200, origin);
    } catch (e: unknown) {
      console.error("Confirm error:", e);
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  // ================== REFERRALS ==================
  if (path === "/api/referrals/stats" && method === "GET") {
    try {
      // Ensure user has referral code
      if (!user.referral_code) {
        const code = generateReferralCode();
        await dbUpdate("users", { telegram_id: telegramId }, { referral_code: code });
        user.referral_code = code;
      }

      const referralCode = user.referral_code as string;
      const stats = await dbSelect("referrals", { referrer_id: telegramId });
      const credits = await dbSelect("referral_credits", { user_id: telegramId });
      const balance = credits[0]?.balance || 0;
      const totalEarned = credits[0]?.total_earned || 0;
      const totalUsed = credits[0]?.total_used || 0;
      const totalCreditsEarned = stats.reduce((sum: number, r: Record<string, unknown>) => sum + parseFloat(r.credits_earned as string || '0'), 0);

      return jsonResponse({
        referralCode,
        referralLink: `https://t.me/${BOT_TOKEN?.split(':')[0]}?start=ref_${referralCode}`,
        totalReferrals: stats.length,
        totalCreditsEarned,
        currentBalance: balance,
        totalUsed,
        referrals: stats,
      }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  // ================== WITHDRAWALS ==================
  if (path === "/api/withdrawals/my" && method === "GET") {
    try {
      const unpaid = await dbSelect("platform_transactions", { channel_owner_id: telegramId, owner_paid_out: false });
      const pendingEarnings = unpaid.reduce((sum: number, r: Record<string, unknown>) => sum + parseFloat(r.owner_share as string || '0'), 0);
      const withdrawals = await dbSelect("withdrawals", { channel_owner_id: telegramId });
      return jsonResponse({ pendingEarnings, withdrawals }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path === "/api/withdrawals/request" && method === "POST") {
    const { amount, wallet_address } = body;
    if (!amount || (amount as number) <= 0) return jsonResponse({ error: "Invalid amount" }, 400, origin);
    if ((amount as number) < MIN_WITHDRAWAL) return jsonResponse({ error: `Minimum withdrawal is ${MIN_WITHDRAWAL} TON` }, 400, origin);
    if (!wallet_address) return jsonResponse({ error: "Wallet address required" }, 400, origin);
    try {
      const unpaid = await dbSelect("platform_transactions", { channel_owner_id: telegramId, owner_paid_out: false });
      const total = unpaid.reduce((s: number, r: Record<string, unknown>) => s + parseFloat(r.owner_share as string || '0'), 0);
      if ((amount as number) > total) return jsonResponse({ error: "Insufficient balance" }, 400, origin);
      await dbInsert("withdrawals", { channel_owner_id: telegramId, amount, wallet_address, status: "pending" });
      return jsonResponse({ success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  // ================== REVIEWS ==================
  if (path === "/api/reviews" && method === "POST") {
    const { channel_id, rating, comment } = body;
    if (!channel_id || !rating || (rating as number) < 1 || (rating as number) > 5) return jsonResponse({ error: "Invalid rating" }, 400, origin);
    try {
      const existing = await dbSelect("reviews", { user_id: telegramId, channel_id: channel_id as string });
      if (existing.length > 0) return jsonResponse({ error: "Already rated" }, 400, origin);
      await dbInsert("reviews", { user_id: telegramId, channel_id, rating, comment: comment || null });
      return jsonResponse({ success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path.match(/^\/api\/reviews\/[a-f0-9-]+$/) && method === "GET") {
    const channelId = path.split("/").pop()!;
    if (!isValidUUID(channelId)) return jsonResponse({ error: "Invalid channel ID" }, 400, origin);
    try {
      const reviews = await dbSelect("reviews", { channel_id: channelId });
      return jsonResponse(reviews.filter((r: Record<string, unknown>) => !r.is_hidden), 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  // ================== REPORTS ==================
  if (path === "/api/reports" && method === "POST") {
    const { channel_id, reason, description } = body;
    if (!channel_id || !reason) return jsonResponse({ error: "Channel and reason required" }, 400, origin);
    try {
      const existing = await dbSelect("reports", { reporter_id: telegramId, reported_channel_id: channel_id as string });
      if (existing.length > 0) return jsonResponse({ error: "Already reported" }, 400, origin);
      await dbInsert("reports", { reporter_id: telegramId, reported_channel_id: channel_id, reason, description: description || null });
      return jsonResponse({ success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  // ================== ADMIN ==================
  if (!isAdmin) return jsonResponse({ error: "Not found" }, 404, origin);

  if (path === "/api/admin/reports" && method === "GET") {
    try {
      const reports = await dbSelect("reports", {});
      const enriched = await Promise.all(reports.map(async (report: Record<string, unknown>) => {
        try {
          const reporters = await dbSelect("users", { telegram_id: report.reporter_id as number });
          const channels = await dbSelect("channels", { id: report.reported_channel_id as string });
          let channelOwner = { first_name: 'Unknown' };
          if (channels[0]?.owner_id) {
            const owners = await dbSelect("users", { telegram_id: channels[0].owner_id as number });
            channelOwner = owners[0] || channelOwner;
          }
          return { ...report, reporter: reporters[0] || { first_name: 'Unknown' }, channel: channels[0] || { channel_name: 'Unknown' }, channel_owner: channelOwner };
        } catch { return report; }
      }));
      return jsonResponse(enriched, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path.match(/^\/api\/admin\/reports\/[a-f0-9-]+\/review$/) && method === "POST") {
    const reportId = path.split("/")[4];
    const { action } = body;
    if (!isValidUUID(reportId)) return jsonResponse({ error: "Invalid report ID" }, 400, origin);
    if (!action || !['ban', 'dismiss'].includes(action as string)) return jsonResponse({ error: "Invalid action" }, 400, origin);
    try {
      const reports = await dbSelect("reports", { id: reportId });
      const report = reports[0];
      if (!report) return jsonResponse({ error: "Report not found" }, 404, origin);
      if (action === 'dismiss') {
        await dbDelete("reports", { id: reportId });
        return jsonResponse({ success: true, action: 'dismissed' }, 200, origin);
      }
      if (action === 'ban') {
        await dbUpdate("reports", { id: reportId }, { status: 'banned', reviewed_by: telegramId, reviewed_at: new Date().toISOString() });
        await dbUpdate("channels", { id: report.reported_channel_id as string }, { is_active: false });
        return jsonResponse({ success: true, action: 'banned' }, 200, origin);
      }
      return jsonResponse({ error: "Invalid action" }, 400, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path === "/api/admin/withdrawals" && method === "GET") {
    try {
      const withdrawals = await dbSelect("withdrawals", { status: 'pending' });
      return jsonResponse(withdrawals, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path.match(/^\/api\/admin\/withdrawals\/[a-f0-9-]+\/approve$/) && method === "POST") {
    const withdrawalId = path.split("/")[4];
    if (!isValidUUID(withdrawalId)) return jsonResponse({ error: "Invalid withdrawal ID" }, 400, origin);
    try {
      const withdrawals = await dbSelect("withdrawals", { id: withdrawalId });
      const withdrawal = withdrawals[0];
      if (!withdrawal || withdrawal.status !== 'pending') return jsonResponse({ error: "Not found or processed" }, 404, origin);

      // FIX: Only mark proportional transactions
      const unpaidTxs = await dbSelect("platform_transactions", { channel_owner_id: withdrawal.channel_owner_id as number, owner_paid_out: false });
      let remaining = parseFloat(withdrawal.amount as string);
      const coveredIds: string[] = [];
      for (const tx of unpaidTxs) {
        if (remaining <= 0) break;
        coveredIds.push(tx.id as string);
        remaining -= parseFloat(tx.owner_share as string);
      }
      for (const txId of coveredIds) {
        await dbUpdate("platform_transactions", { id: txId }, { owner_paid_out: true, withdrawal_id: withdrawalId });
      }

      await dbUpdate("withdrawals", { id: withdrawalId }, { status: 'processed', processed_date: new Date().toISOString(), processed_by: telegramId });
      return jsonResponse({ success: true, covered: coveredIds.length }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path === "/api/admin/verifications" && method === "GET") {
    try {
      const requests = await dbSelect("verification_requests", { status: 'pending' });
      return jsonResponse(requests, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  if (path.match(/^\/api\/admin\/verifications\/[a-f0-9-]+\/review$/) && method === "POST") {
    const requestId = path.split("/")[4];
    const { action, notes } = body;
    if (!isValidUUID(requestId)) return jsonResponse({ error: "Invalid request ID" }, 400, origin);
    try {
      const requests = await dbSelect("verification_requests", { id: requestId });
      const request = requests[0];
      if (!request || request.status !== 'pending') return jsonResponse({ error: "Not found or processed" }, 404, origin);
      await dbUpdate("verification_requests", { id: requestId }, {
        status: action === 'approve' ? 'approved' : 'rejected',
        reviewed_by: telegramId,
        reviewed_at: new Date().toISOString(),
        admin_notes: notes || null,
      });
      if (action === 'approve') {
        await dbUpdate("channels", { id: request.channel_id as string }, { is_verified: true, verified_at: new Date().toISOString() });
      }
      return jsonResponse({ success: true }, 200, origin);
    } catch (e: unknown) {
      return jsonResponse({ error: e instanceof Error ? e.message : 'Failed' }, 500, origin);
    }
  }

  return jsonResponse({ error: "Not found" }, 404, origin);
}

// ================== MAIN SERVER ==================
serve(async (req: Request) => {
  const url = new URL(req.url);
  const path = url.pathname;
  const method = req.method;
  const origin = req.headers.get("Origin");

  if (method === "OPTIONS") return new Response(null, { headers: getCorsHeaders(origin) });

  // Health check
  if (path === "/health" || path === "/") {
    try {
      const { error } = await supabase.from('users').select('telegram_id').limit(1);
      if (error) throw error;
      return jsonResponse({ status: 'healthy', version: '2.2.0', timestamp: new Date().toISOString(), network: TON_NETWORK }, 200, origin);
    } catch {
      return jsonResponse({ status: 'unhealthy', error: 'Database connection failed' }, 503, origin);
    }
  }

  // Bot webhook
  if (path.endsWith("/bot-webhook") && method === "POST") {
    if (!verifyWebhookSecret(req.headers)) return new Response("Unauthorized", { status: 401, headers: getCorsHeaders(origin) });
    try {
      const body = await req.json();
      await handleBotUpdate(body);
      return new Response("OK", { headers: getCorsHeaders(origin) });
    } catch (e) {
      console.error("Bot webhook error:", e);
      return new Response("Error", { status: 500, headers: getCorsHeaders(origin) });
    }
  }

  // API routes
  if (path.includes("/api/")) {
    try {
      const apiPath = path.substring(path.indexOf("/api/"));
      let body: Record<string, unknown> = {};
      if (["POST", "PUT", "PATCH"].includes(method)) {
        try { body = await req.json(); } catch { return jsonResponse({ error: "Invalid JSON" }, 400, origin); }
      }
      return await apiHandler(apiPath, method, req.headers, body, origin);
    } catch (e) {
      console.error("API error:", e);
      return jsonResponse({ error: "Internal server error" }, 500, origin);
    }
  }

  return new Response("MySubHub API v2.2", { headers: getCorsHeaders(origin) });
});
